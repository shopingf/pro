رابط تطبيق الويب (Google Apps Script URL)

ضع رابط السكريبت الخاص بك لتسجيل كل طلبية جديدة تلقائياً في شيت غوغل

جاهز للربط اضغط على زر الاختبار للتحقق من الاتصال المباشر.

طريقة الربط السريعة في 60 ثانية:

  1. أنشئ جدول بيانات جديد في Google Sheets.
  2. من القائمة العلوية اختر Extensions > Apps Script.
  3. امسح الكود القديم والصق كود السكريبت الموجود بالجانب.
  4. اضغط Deploy > New deployment واختر Web app.
  5. اجعل الصلاحية: Anyone ثم انقر Deploy.
  6. انسخ الرابط المولد وضعه في الحقل أعلاه واضغط حفظ الرابط.

كود Google Apps Script الجاهز

انسخ هذا الكود بالكامل وضعه في محرر Apps Script

/**
 * GRIS BLANC™ BOUTIQUE - GOOGLE APPS SCRIPT ORDER SYNC
 */
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Order ID", "Date", "Time", "Customer Name", "Phone Number",
        "Wilaya", "Commune", "Product Variant", "Color", "Size",
        "Quantity", "Total Price", "Notes", "User Agent", "Status"
      ]);
      sheet.getRange(1, 1, 1, 15).setFontWeight("bold").setBackground("#C5A059").setFontColor("#FFFFFF");
    }
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try { data = JSON.parse(e.postData.contents); } catch (err) { data = e.parameter || {}; }
    } else if (e && e.parameter) { data = e.parameter; }
    sheet.appendRow([
      data.orderId || "GB-" + Math.floor(10000 + Math.random() * 90000),
      data.date || new Date().toLocaleDateString('ar-DZ'),
      data.time || new Date().toLocaleTimeString('ar-DZ'),
      data.fullname || "N/A", data.phone || "N/A",
      data.wilaya || "N/A", data.commune || "N/A",
      data.variant || "Gris Blanc™ Edition", data.color || "Blanc", data.size || "42",
      data.quantity || 1, data.price || "4,500 DZD",
      data.notes || "", data.userAgent || "", "NEW_ORDER_COD"
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", result: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally { lock.releaseLock(); }
}
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ status: "active", message: "Gris Blanc Webhook Active" })).setMimeType(ContentService.MimeType.JSON);
}

إعدادات ربط Google Sheets & Apps Script

تجهيز واستقبال طلبات الزبائن مباشرة في جدول بيانات غوغل شيت الخاص بك

جاهز للاختبار اضغط على اختبار الاتصال للتأكد من استقبال السكريبت للطلبات

💡 طريقة الربط الخطوة بخطوة (خلال 60 ثانية فقط):

  1. أنشئ جدول بيانات جديد في Google Sheets لجمع طلبات الزبائن.
  2. من القائمة العلوية اضغط على: Extensions (الإضافات) > Apps Script.
  3. افتح تبويب "كود Google Apps Script" من الأعلى، وانسخ الكود بنقرة واحدة.
  4. في محرر Apps Script امسح الكود الافتراضي والصق الكود المنسوخ، ثم اضغط Deploy > New deployment.
  5. اختر Web app واجعل الصلاحية: Anyone (أي شخص) واضغط Deploy.
  6. انسخ رابط Web App URL المولد وضعه في الحقل أعلاه واضغط "حفظ الرابط".
